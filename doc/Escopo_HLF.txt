### 3. ARQUIVO: assets/js/bloco-protecao.js

```javascript
/**
 * HLF Protection Module v1.1
 * Selective protection respecting accessibility
 */

const HLF_PROTECTION = {
    init() {
        if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

        this.protectImages();
        this.injectCSS();
    },

    protectImages() {
        document.addEventListener('contextmenu', (e) => {
            if (e.target.tagName === 'IMG' || e.target.tagName === 'FIGURE') {
                e.preventDefault();
            }
        }, false);
    },

    injectCSS() {
        const style = document.createElement('style');
        style.innerHTML = `
            [data-protect] {
                -webkit-user-select: none;
                -moz-user-select: none;
                -ms-user-select: none;
                user-select: none;
            }
            img { pointer-events: none; }
        `;
        document.head.appendChild(style);
    }
};

HLF_PROTECTION.init();
```

const CALC_KIT = {
    custoRefeicao(produtosIds) {
        let totalPreco = 0;
        let totalPV = 0;
        
        produtosIds.forEach(id => {
            const p = PRODUTOS_HLF.find(item => item.id === id);
            if (p) {
                totalPreco += p.precoSugerido;
                totalPV += p.pontosVolume;

            custo: custoPorDose,
